cmake_minimum_required(VERSION 3.16)
project(convoy CXX)
set(CMAKE_CXX_STANDARD 17)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(nav_msgs REQUIRED)
find_package(std_msgs REQUIRED)
# Same headers as the ESP32 firmware and the host tests: one source of truth for packet + watchdog logic.
include_directories(${CMAKE_CURRENT_SOURCE_DIR}/../../../firmware/lib/convoy)
add_executable(convoy_follower_node src/convoy_follower_node.cpp)
ament_target_dependencies(convoy_follower_node rclcpp geometry_msgs nav_msgs std_msgs)
add_executable(convoy_lead_node src/convoy_lead_node.cpp)
ament_target_dependencies(convoy_lead_node rclcpp nav_msgs)
install(TARGETS convoy_follower_node convoy_lead_node DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY launch DESTINATION share/${PROJECT_NAME})
ament_package()
